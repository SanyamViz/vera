<script>
        function copyNotice() {
            const text = document.getElementById('notice-text').innerText;
            navigator.clipboard.writeText(text).then(() => {
                const btn = event.currentTarget;
                const originalText = btn.innerHTML;
                btn.innerHTML = `<span class="material-symbols-outlined text-sm" data-icon="check">check</span> Copied!`;
                btn.classList.add('bg-primary-fixed', 'text-on-primary-fixed');
                btn.classList.remove('bg-primary', 'text-on-primary');
                
                setTimeout(() => {
                    btn.innerHTML = originalText;
                    btn.classList.remove('bg-primary-fixed', 'text-on-primary-fixed');
                    btn.classList.add('bg-primary', 'text-on-primary');
                }, 2000);
            }).catch(err => {
                console.error('Failed to copy text: ', err);
            });
        }
    </script>
